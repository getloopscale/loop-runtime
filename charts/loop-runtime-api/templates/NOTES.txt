LoopRuntime API has been deployed.

1. Get the application URL:
{{- if eq .Values.service.type "ClusterIP" }}
  kubectl port-forward svc/{{ include "loop-runtime-common.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }} -n {{ .Release.Namespace }}
  echo "API: http://localhost:{{ .Values.service.port }}/api/v1/health"
{{- else if eq .Values.service.type "NodePort" }}
  export NODE_PORT=$(kubectl get svc {{ include "loop-runtime-common.fullname" . }} -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[?(@.name=="http")].nodePort}')
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="InternalIP")].address}')
  echo "API: http://$NODE_IP:$NODE_PORT/api/v1/health"
{{- end }}

2. Verify the deployment:
  kubectl get pods -l "app.kubernetes.io/name={{ include "loop-runtime-common.name" . }}" -n {{ .Release.Namespace }}

3. Run the Helm test:
  helm test {{ .Release.Name }} -n {{ .Release.Namespace }}
